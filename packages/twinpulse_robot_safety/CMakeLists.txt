cmake_minimum_required(VERSION 3.16)
project(twinpulse_robot_safety LANGUAGES CXX)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME deployment_runtime COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_test.py)
add_test(NAME stationary_lifecycle COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/stationary_test.py)
add_test(NAME manual_lifecycle COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/teleop_test.py)
add_test(NAME manual_stop_policy COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/stop_policy_test.py)
add_test(NAME lidar_capture COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_lidar_capture.py)
add_test(NAME isolated_driver_identity COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_isolated_ros_scan.py)
add_test(NAME robot_journal COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_robot_journal.py)
add_executable(command_guard_test tests/command_guard_test.cpp)
target_include_directories(command_guard_test PRIVATE include)
target_compile_features(command_guard_test PRIVATE cxx_std_17)
target_compile_options(command_guard_test PRIVATE -Wall -Wextra -Werror -pedantic)
add_test(NAME command_guard COMMAND command_guard_test)
add_executable(torque_off_test tests/torque_off_test.cpp)
target_compile_features(torque_off_test PRIVATE cxx_std_17)
target_compile_options(torque_off_test PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
add_test(NAME torque_off COMMAND torque_off_test)
option(BUILD_ROS_TESTS "Build fake-SDK ROS integration test" OFF)
if(BUILD_ROS_TESTS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  add_executable(ros_guard_test tests/ros_guard_test.cpp)
  target_include_directories(ros_guard_test BEFORE PRIVATE tests/fake_include include)
  target_compile_features(ros_guard_test PRIVATE cxx_std_17)
  target_compile_options(ros_guard_test PRIVATE -Wall -Wextra -Werror -pedantic)
  ament_target_dependencies(ros_guard_test rclcpp geometry_msgs std_msgs std_srvs)
  add_test(NAME ros_guard COMMAND ros_guard_test)
  set_tests_properties(ros_guard PROPERTIES TIMEOUT 30)
endif()
